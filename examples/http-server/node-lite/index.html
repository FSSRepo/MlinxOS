Node Lite combina QuickJS con una capa compacta de compatibilidad Node para ejecutar scripts y levantar servicios HTTP en sistemas ligeros.

server.js
// Un servidor pequeño, una idea clara
const http = require('http');

http.createServer((req, res) => {
  res.writeHead(200, {
    'content-type': 'text/plain'
  });
  res.end('Hola desde Node Lite');
}).listen(8400);

$ node-lite server.js
