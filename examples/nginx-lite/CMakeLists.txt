cmake_minimum_required(VERSION 3.16)
project(nginx_lite LANGUAGES C CXX)

option(NGINX_LITE_TLS "Enable HTTPS with the bundled mbedTLS sources" ON)
add_executable(nginx-lite src/main.cpp)
target_compile_features(nginx-lite PRIVATE cxx_std_17)
if(NGINX_LITE_TLS)
    set(MBEDTLS_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../external/mbedtls)
    file(GLOB MBEDTLS_SOURCES ${MBEDTLS_DIR}/library/*.c)
    add_library(nginx_lite_mbedtls STATIC ${MBEDTLS_SOURCES})
    target_include_directories(nginx_lite_mbedtls PUBLIC ${MBEDTLS_DIR}/include)
    if(MSVC)
        target_compile_options(nginx_lite_mbedtls PRIVATE /w)
    else()
        target_compile_options(nginx_lite_mbedtls PRIVATE -w)
    endif()
    target_compile_definitions(nginx-lite PRIVATE NGINX_LITE_TLS=1)
    target_link_libraries(nginx-lite PRIVATE nginx_lite_mbedtls)
    add_executable(nginx-lite-cert src/make_cert.c)
    target_link_libraries(nginx-lite-cert PRIVATE nginx_lite_mbedtls)
    if(WIN32)
        target_link_libraries(nginx-lite PRIVATE bcrypt ws2_32)
        target_link_libraries(nginx-lite-cert PRIVATE bcrypt)
    endif()
endif()
if(WIN32)
    target_link_libraries(nginx-lite PRIVATE ws2_32)
else()
    target_compile_definitions(nginx-lite PRIVATE _POSIX_C_SOURCE=200809L)
endif()
install(TARGETS nginx-lite RUNTIME DESTINATION bin)
